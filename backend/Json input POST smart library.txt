#perintah peminjaman

DECLARE
    v_member_id NUMBER;
    v_staff_id  NUMBER;
    v_copy_id   NUMBER;
    v_loan_id   NUMBER;
BEGIN
    -- Ambil ID anggota
    SELECT member_id
    INTO v_member_id
    FROM members
    WHERE email = 'testmember@library.com';

    -- Ambil ID petugas
    SELECT staff_id
    INTO v_staff_id
    FROM staff
    WHERE username = 'teststaff';

    -- Ambil ID eksemplar
    SELECT copy_id
    INTO v_copy_id
    FROM book_copies
    WHERE barcode = 'CC-TEST-001';

    -- Jalankan Procedure peminjaman
    sp_borrow_book(
        p_member_id => v_member_id,
        p_staff_id  => v_staff_id,
        p_copy_id   => v_copy_id,
        p_due_date  => TRUNC(SYSDATE) + 7,
        p_loan_id   => v_loan_id
    );

    COMMIT;
END;

SELECT
    loan_id,
    member_id,
    staff_id,
    copy_id,
    loan_date,
    due_date,
    return_date,
    status
FROM loans
ORDER BY loan_id DESC;

#periksa tenggat
SELECT
    l.loan_id,
    l.due_date,
    l.status,
    b.title,
    m.name
FROM loans l
JOIN book_copies bc ON l.copy_id = bc.copy_id
JOIN books b ON bc.book_id = b.book_id
JOIN members m ON l.member_id = m.member_id
WHERE bc.barcode = 'CC-TEST-001';
